51 public function evaluate(Request $request, Postulation $postulation)
52 {
53 $user = Auth::user();
54 $professor = Professor::where('user_id', $user->id)->firstOrFail();
55
56 // Validar permisos de city_program
57 if ($postulation->leadStudent->city_program_id !== $professor->city_program_id) {
58 abort(403, 'No tienes permiso para evaluar esta postulación.');
59 }
60
61 $request->validate([
62 'status' => 'required|in:approved,rejected',
63 'review_comment' => 'required|string|max:1000',
64 ]);
65
66 DB::beginTransaction();
67 try {
68 // Serializar las aprobaciones de un mismo proyecto para que dos
69 // evaluaciones simultáneas no puedan reemplazar la asignación.
70 $postulation = Postulation::whereKey($postulation->getKey())
71 ->lockForUpdate()
72 ->firstOrFail();
73
74 if ($postulation->status !== 'pending') {
75 DB::rollBack();
76
77 return back()->with('error', 'Esta postulación ya ha sido evaluada.');
78 }
79
80 $project = Project::whereKey($postulation->project_id)
81 ->lockForUpdate()
82 ->firstOrFail();
83
84 if ($request->status === 'approved') {
85 $assignedStatusId = ProjectStatus::where('name', 'Asignado')->value('id');
86 $alreadyAssigned = $project->approved_postulation_id !== null
87 || ($assignedStatusId !== null && (int) $project->project_status_id === (int) $assignedStatusId);
88
89 if ($alreadyAssigned) {
90 DB::rollBack();
91
92 return back()->with(
93 'error',
94 'No se puede aprobar esta postulación: el proyecto ya fue asignado a otra postulación.'
95 );
96 }
97
98 if (! $project->isAvailableForStudent($postulation->leadStudent)) {
99 DB::rollBack();
100
101 return back()->with(
102 'error',
103 'No se puede aprobar esta postulación: el proyecto no está disponible para postulación (debe estar "Aprobado" y pertenecer al programa del estudiante).'
104 );
105 }
106 }
107
108 $postulation->update([
109 'status' => $request->status,
110 'review_comment' => $request->review_comment,
111 ]);
112
113 // Si se rechaza, liberar la prioridad
114 if ($request->status === 'rejected') {
115 $this->cleanupPriority($postulation);
116 }
117
118 if ($request->status === 'approved') {
119 $activePeriod = AcademicCalendarService::currentActivePeriodOrFail();
120 $assignedStatusId = ProjectStatus::where('name', 'Asignado')->firstOrFail()->id;
121
122 // 1. Asignar proyecto y vincular con la postulación aprobada
123 $project->update([
124 'project_status_id' => $assignedStatusId,
125 'approved_postulation_id' => $postulation->postulation_id,
126 'assignment_academic_period_id' => $activePeriod->id,
127 'assigned_at' => now(),
128 ]);
129
130 // 2. Sincronizar estudiantes con la referencia de la postulación
131 $studentIds = $postulation->members->pluck('student_id')->toArray();
132
133 // Usamos sync con datos adicionales para el pivot si es necesario,
134 // o simplemente adjuntamos la postulation_id
135 $syncData = [];
136 foreach ($studentIds as $id) {
137 $syncData[$id] = ['postulation_id' => $postulation->postulation_id];
138 }
139 $project->students()->sync($syncData);
140
141 // 3. Cancelar otras postulaciones para TODOS los integrantes involucrados
142 foreach ($studentIds as $studentId) {
143 $otherPostulations = Postulation::where('status', 'pending')
144 ->where('postulation_id', '!=', $postulation->postulation_id)
145 ->whereHas('members', function ($q) use ($studentId) {
146 $q->where('student_id', $studentId);
147 })
148 ->get();
149
150 foreach ($otherPostulations as $other) {
151 $other->update([
152 'status' => 'rejected',
153 'review_comment' => 'Cancelada automáticamente por aprobación de otra postulación.',
154 ]);
155 // Liberar prioridades de las postulaciones canceladas
156 $this->cleanupPriority($other);
157 }
158 }
159
160 AcademicCalendarService::recordProjectStage(
161 $project,
162 'assigned',
163 $activePeriod,
164 Auth::id(),
165 'Proyecto asignado vía aprobación de postulación. '.$request->review_comment,
166 ['postulation_id' => $postulation->postulation_id, 'student_ids' => $studentIds]
167 );
168 }
169
170 DB::commit();
171
172 return redirect()->route('projects.evaluation.postulations.index')
173 ->with('success', 'Postulación evaluada correctamente.');
174
175 } catch (\Exception $e) {
176 DB::rollBack();
177
178 // Guardamos el error real en el log (solo lo ven los programadores)
179 \Log::error('Error al evaluar la postulación: ' . $e->getMessage(), ['exception' => $e]);
180
181 // Al comité le mostramos un mensaje amigable sin el SQL
182 return back()->with('error', 'Ocurrió un error inesperado al evaluar la postulación. Por favor, inténtalo de nuevo.');
183 }

188 private function cleanupPriority(Postulation $postulation): void
189 {
190 $priorityRecord = PostulationPriority::where('postulation_id', $postulation->postulation_id)
191 ->first();
192
193 if ($priorityRecord) {
194 $studentId = $priorityRecord->student_id;
195 $order = $priorityRecord->priority_order;
196
197 // Eliminar el registro de prioridad actual
198 $priorityRecord->delete();
199
200 // Reordenar las prioridades superiores del mismo estudiante para cerrar el hueco
201 PostulationPriority::where('student_id', $studentId)
202 ->where('priority_order', '>', $order)
203 ->decrement('priority_order');
204 }
