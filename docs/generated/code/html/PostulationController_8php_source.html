295 public function destroy(Postulation $postulation)
296 {
297 $student = Student::where('user_id', Auth::id())->firstOrFail();
298
299 // Solo el líder que creó la postulación puede cancelarla
300 if ($postulation->lead_student_id !== $student->id) {
301 abort(403, 'Solo el líder que creó la postulación puede cancelarla.');
302 }
303
304 if ($postulation->status !== 'pending') {
305 return back()->with('error', 'Solo puedes cancelar postulaciones pendientes.');
306 }
307
308 DB::beginTransaction();
309 try {
310 // Reordenar prioridades después de la eliminación
311 $priorityToDelete = PostulationPriority::where('student_id', $student->id)
312 ->where('postulation_id', $postulation->postulation_id)
313 ->first();
314
315 if ($priorityToDelete) {
316 $priorityOrder = $priorityToDelete->priority_order;
317 $priorityToDelete->delete();
318
319 // Liberar el espacio de prioridad para este estudiante
320 PostulationPriority::where('student_id', $student->id)
321 ->where('priority_order', '>', $priorityOrder)
322 ->decrement('priority_order');
323 }
324
325 // Eliminar archivo de notas si existe
326 if ($postulation->grades_file && Storage::disk('local')->exists($postulation->grades_file)) {
327 Storage::disk('local')->delete($postulation->grades_file);
328 }
329
330 // Al eliminar la postulación, Eloquent se encarga de los miembros si hay cascade en la BD
331 // o los eliminamos manualmente para mayor seguridad
332 $postulation->delete();
333
334 DB::commit();
335
336 return back()->with('success', 'Postulación cancelada correctamente y prioridad liberada.');
337 } catch (\Exception $e) {
338 DB::rollBack();
339
340 return back()->with('error', 'Error al cancelar la postulación: ' . $e->getMessage());
341 }
