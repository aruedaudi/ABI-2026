81 public function evaluate(Request $request, Project $project)
82 {
83 $committeeLeader = $this->resolveCommitteeLeader();
84 $this->ensureProjectBelongsToCommitteeProgram($project, $committeeLeader->city_program_id);
85 $validated = $request->validate([
86 'status' => 'required|string|in:Aprobado,Rechazado,Devuelto para correccion,Devuelto para corrección',
87 'comments' => 'nullable|string',
88 ]);
89
90 $comments = $this->normalizeComments($validated['comments'] ?? null);
91 $requestedStatus = Str::of($validated['status'])->ascii()->squish()->toString();
92 $statusName = $requestedStatus;
93 $isProfessorProject = $project->professors()->exists();
94 $isStudentProject = ! $isProfessorProject;
95
96 // Si es proyecto de estudiante y se aprueba, normalmente se asigna.
97 // Si estamos en una fase distinta, este comportamiento se puede ajustar.
98 // Sin embargo, si el estudiante propuso la idea, ya es suya.
99 if ($statusName === 'Aprobado' && $isStudentProject) {
100 $statusName = 'Asignado';
101 }
102
103 $normalizedStatusName = Str::of($statusName)->ascii()->toString();
104 $status = ProjectStatus::whereIn('name', array_unique([$statusName, $normalizedStatusName]))->first();
105 if (! $status) {
106 return back()->with('error', "No se encontro el estado '$statusName'.");
107 }
108
109 $project->update(['project_status_id' => $status->id]);
110
111 if (in_array($statusName, ['Devuelto para correccion'], true)) {
112 $latestVersion = $project->versions()->latest('created_at')->first();
113 if ($latestVersion) {
114 $commentContent = Content::where('name', 'Comentarios')->whereJsonContains('roles', 'committee_leader')->first();
115 if ($commentContent) {
116 ContentVersion::create([
117 'version_id' => $latestVersion->id,
118 'content_id' => $commentContent->id,
119 'value' => $comments ?? 'Sin comentarios',
120 ]);
121 }
122 }
123 }
124
125 $activePeriod = \App\Models\AcademicPeriod::query()->active()->first() ?? $project->proposalAcademicPeriod;
126 $normalizedRequestedStatus = Str::of($validated['status'])->ascii()->lower()->toString();
127 $stage = match ($normalizedRequestedStatus) {
128 'aprobado' => 'approved',
129 'rechazado' => 'rejected',
130 'devuelto para correccion' => 'returned_for_correction',
131 default => 'evaluated',
132 };
133
134 AcademicCalendarService::recordProjectStage(
135 $project,
136 $stage,
137 $activePeriod,
138 Auth::id(),
139 $comments,
140 ['final_status_name' => $statusName]
141 );
142
143 // Disparar evento de notificacion
144 event(new ProjectIdeaEvaluated(
145 $project->load(['students.user', 'professors.user']),
146 $statusName,
147 $comments
148 ));
149
150 return redirect()
151 ->route('projects.evaluation.index')
152 ->with('success', "Evaluacion del proyecto '{$project->title}' enviada correctamente con estado: $statusName.");
153 }

495 protected function publicAssetDataUri(string $relativePath): ?string
496 {
497 $absolutePath = public_path($relativePath);
498
499 if (! is_file($absolutePath)) {
500 return null;
501 }
502
503 $mimeType = mime_content_type($absolutePath) ?: 'image/png';
504 $contents = file_get_contents($absolutePath);
505
506 if ($contents === false) {
507 return null;
508 }
509
510 return 'data:' . $mimeType . ';base64,' . base64_encode($contents);
