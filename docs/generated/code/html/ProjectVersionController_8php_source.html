137 protected function authorizeHistoryAccess(Project $project, ?User $user): void
138 {
139 if (! $user) {
140 abort(403);
141 }
142
143 if ($user->role === 'research_staff') {
144 return;
145 }
146
147 $project->loadMissing(['professors', 'students']);
148
149 if (in_array($user->role, ['professor', 'committee_leader'], true)) {
150 $professor = $user->professor ?? Professor::query()->where('user_id', $user->id)->first();
151
152 if ($professor && $project->professors->contains('id', $professor->id)) {

156 if ($user->role === 'committee_leader' && $professor && $professor->committee_leader && $professor->city_program_id) {

160 if ($belongsToProgram) {
161 return;
162 }

164
165 abort(403, 'No estás asignado a este proyecto.');
166 }
167
168 if ($user->role === 'student') {
169 $student = $user->student;
